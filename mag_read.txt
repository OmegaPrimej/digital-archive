# ===== COMPLETE INTERFACE DISCOVERY =====

class UniversalInterfaceScanner:
    """Discover ALL available communication interfaces on system"""
    
    def __init__(self):
        self.interfaces = defaultdict(list)
        self.scan_all()
    
    def scan_all(self) -> Dict[InterfaceType, List[str]]:
        """Scan for every possible interface"""
        print("🔍 Scanning ALL communication interfaces...")
        
        # Ethernet interfaces
        self.scan_ethernet()
        
        # WiFi interfaces
        self.scan_wifi()
        
        # Bluetooth
        self.scan_bluetooth()
        
        # USB devices
        self.scan_usb()
        
        # Serial ports
        self.scan_serial()
        
        # Audio devices
        self.scan_audio()
        
        # Special interfaces
        self.scan_special()
        
        print(f"✅ Found {sum(len(v) for v in self.interfaces.values())} interfaces")
        return self.interfaces
    
    def scan_ethernet(self):
        """Scan Ethernet interfaces"""
        try:
            import netifaces
            for iface in netifaces.interfaces():
                if iface.startswith(('eth', 'en', 'em')):
                    self.interfaces[InterfaceType.ETHERNET].append(iface)
                    # Get details
                    addrs = netifaces.ifaddresses(iface)
                    if netifaces.AF_LINK in addrs:
                        mac = addrs[netifaces.AF_LINK][0]['addr']
                        self.interfaces[InterfaceType.ETHERNET].append(f"{iface} ({mac})")
        except:
            # Fallback to /sys/class/net
            try:
                net_path = "/sys/class/net"
                if os.path.exists(net_path):
                    for iface in os.listdir(net_path):
                        if iface.startswith(('eth', 'en', 'em')):
                            self.interfaces[InterfaceType.ETHERNET].append(iface)
            except:
                pass
    
    def scan_wifi(self):
        """Scan WiFi interfaces"""
        try:
            # Using iw
            result = subprocess.run(['iw', 'dev'], capture_output=True, text=True)
            for line in result.stdout.split('\n'):
                if 'Interface' in line:
                    iface = line.split()[1]
                    self.interfaces[InterfaceType.WIFI].append(iface)
        except:
            # Check for wlan interfaces
            net_path = "/sys/class/net"
            if os.path.exists(net_path):
                for iface in os.listdir(net_path):
                    if iface.startswith('wlan'):
                        self.interfaces[InterfaceType.WIFI].append(iface)
    
    def scan_bluetooth(self):
        """Scan Bluetooth devices"""
        try:
            result = subprocess.run(['hcitool', 'dev'], capture_output=True, text=True)
            for line in result.stdout.split('\n'):
                if 'hci' in line:
                    parts = line.split()
                    if len(parts) >= 2:
                        self.interfaces[InterfaceType.BLUETOOTH].append(parts[0])
        except:
            # Check Bluetooth sysfs
            bt_path = "/sys/class/bluetooth"
            if os.path.exists(bt_path):
                for dev in os.listdir(bt_path):
                    self.interfaces[InterfaceType.BLUETOOTH].append(dev)
    
    def scan_usb(self):
        """Scan USB devices"""
        try:
            usb_path = "/sys/bus/usb/devices"
            if os.path.exists(usb_path):
                for dev in os.listdir(usb_path):
                    if dev.startswith(('usb', '1-', '2-', '3-', '4-')):
                        try:
                            vendor_file = os.path.join(usb_path, dev, 'idVendor')
                            product_file = os.path.join(usb_path, dev, 'idProduct')
                            if os.path.exists(vendor_file):
                                with open(vendor_file, 'r') as f:
                                    vendor = f.read().strip()
                                with open(product_file, 'r') as f:
                                    product = f.read().strip()
                                dev_str = f"usb:{vendor}:{product}"
                                self.interfaces[InterfaceType.USB].append(dev_str)
                        except:
                            pass
        except:
            pass
    
    def scan_serial(self):
        """Scan serial ports"""
        try:
            # Check /dev for serial devices
            dev_path = "/dev"
            serial_patterns = ['ttyUSB', 'ttyACM', 'ttyAMA', 'ttyS', 'tty.']
            for dev in os.listdir(dev_path):
                for pattern in serial_patterns:
                    if dev.startswith(pattern):
                        full_path = os.path.join(dev_path, dev)
                        self.interfaces[InterfaceType.SERIAL].append(full_path)
        except:
            pass
    
    def scan_audio(self):
        """Scan audio interfaces"""
        try:
            # ALSA devices
            result = subprocess.run(['aplay', '-l'], capture_output=True, text=True)
            for line in result.stdout.split('\n'):
                if 'card' in line and 'device' in line:
                    self.interfaces[InterfaceType.AUDIO].append(line.strip())
            
            # PulseAudio
            result = subprocess.run(['pactl', 'list', 'short', 'sinks'], 
                                  capture_output=True, text=True)
            for line in result.stdout.split('\n'):
                if line:
                    self.interfaces[InterfaceType.AUDIO].append(f"pulse:{line.split()[1]}")
        except:
            pass
    
    def scan_special(self):
        """Scan special interfaces"""
        # CAN bus
        can_path = "/sys/class/net"
        if os.path.exists(can_path):
            for iface in os.listdir(can_path):
                if iface.startswith('can'):
                    self.interfaces[InterfaceType.CAN_BUS].append(iface)
        
        # I2C
        i2c_path = "/sys/bus/i2c/devices"
        if os.path.exists(i2c_path):
            for dev in os.listdir(i2c_path):
                self.interfaces[InterfaceType.I2C].append(dev)
        
        # SPI
        spi_path = "/sys/bus/spi/devices"
        if os.path.exists(spi_path):
            for dev in os.listdir(spi_path):
                self.interfaces[InterfaceType.SPI].append(dev)
        
        # GPIO (Raspberry Pi)
        if os.path.exists("/sys/class/gpio"):
            self.interfaces[InterfaceType.GPIO].append("gpiochip0")
    
    def print_all_interfaces(self):
        """Print all discovered interfaces"""
        print("\n" + "="*80)
        print("🌐 COMPLETE INTERFACE INVENTORY")
        print("="*80)
        
        for iface_type, devices in self.interfaces.items():
            if devices:
                print(f"\n{iface_type.value.upper()}:")
                for dev in devices[:10]:  # Show first 10
                    print(f"  • {dev}")
                if len(devices) > 10:
                    print(f"  ... and {len(devices) - 10} more")
        
        print("\n" + "="*80)
    
    def get_interface_by_type(self, iface_type: InterfaceType) -> List[str]:
        """Get interfaces of specific type"""
        return self.interfaces.get(iface_type, [])

# ===== UNIVERSAL INTERFACE CONTROL =====

class PhantomInterfaceController:
    """
    Control ANY communication interface at the hardware level
    """
    
    def __init__(self):
        self.scanner = UniversalInterfaceScanner()
        self.active_interfaces = {}
        self.interface_handlers = {}
        self.setup_handlers()
    
    def setup_handlers(self):
        """Setup interface-specific handlers"""
        self.interface_handlers = {
            InterfaceType.ETHERNET: self._handle_ethernet,
            InterfaceType.WIFI: self._handle_wifi,
            InterfaceType.BLUETOOTH: self._handle_bluetooth,
            InterfaceType.USB: self._handle_usb,
            InterfaceType.SERIAL: self._handle_serial,
            InterfaceType.AUDIO: self._handle_audio,
            InterfaceType.MAGNETIC_STRIPE: self._handle_magnetic_stripe,
            InterfaceType.CAN_BUS: self._handle_can_bus,
            InterfaceType.I2C: self._handle_i2c,
            InterfaceType.SPI: self._handle_spi,
            InterfaceType.GPIO: self._handle_gpio,
        }
    
    def open_interface(self, iface_type: InterfaceType, device: str) -> bool:
        """Open a specific interface"""
        print(f"🔓 Opening {iface_type.value}: {device}")
        
        try:
            if iface_type in self.interface_handlers:
                handler = self.interface_handlers[iface_type]
                result = handler('open', device)
                
                if result:
                    self.active_interfaces[device] = {
                        'type': iface_type,
                        'handle': result,
                        'opened': time.time()
                    }
                    return True
            return False
            
        except Exception as e:
            print(f"❌ Failed to open {device}: {e}")
            return False
    
    def send_data(self, device: str, data: bytes) -> bool:
        """Send data through specific interface"""
        if device not in self.active_interfaces:
            print(f"❌ Interface {device} not opened")
            return False
        
        iface_info = self.active_interfaces[device]
        iface_type = iface_info['type']
        
        if iface_type in self.interface_handlers:
            handler = self.interface_handlers[iface_type]
            return handler('send', device, data)
        
        return False
    
    def receive_data(self, device: str, timeout: float = 1.0) -> Optional[bytes]:
        """Receive data from interface"""
        if device not in self.active_interfaces:
            return None
        
        iface_info = self.active_interfaces[device]
        iface_type = iface_info['type']
        
        if iface_type in self.interface_handlers:
            handler = self.interface_handlers[iface_type]
            return handler('receive', device, timeout)
        
        return None
    
    # === INTERFACE-SPECIFIC HANDLERS ===
    
    def _handle_ethernet(self, action: str, *args):
        """Handle Ethernet interface"""
        if action == 'open':
            device = args[0]
            # Create raw socket
            sock = socket.socket(socket.AF_PACKET, socket.SOCK_RAW)
            try:
                sock.bind((device, 0))
                return sock
            except:
                return None
        
        elif action == 'send':
            device, data = args
            sock = self.active_interfaces[device]['handle']
            return sock.send(data) > 0
        
        elif action == 'receive':
            device, timeout = args
            sock = self.active_interfaces[device]['handle']
            sock.settimeout(timeout)
            try:
                return sock.recv(65536)
            except socket.timeout:
                return None
    
    def _handle_wifi(self, action: str, *args):
        """Handle WiFi interface"""
        # Similar to Ethernet but with additional WiFi headers
        return self._handle_ethernet(action, *args)
    
    def _handle_bluetooth(self, action: str, *args):
        """Handle Bluetooth interface"""
        if action == 'open':
            device = args[0]  # e.g., 'hci0'
            try:
                import bluetooth
                sock = bluetooth.BluetoothSocket(bluetooth.RAW)
                sock.bind((device, 0))
                return sock
            except ImportError:
                print("⚠️  Install pybluez: pip install pybluez")
                return None
        
        elif action == 'send':
            device, data = args
            sock = self.active_interfaces[device]['handle']
            return sock.send(data) > 0
    
    def _handle_usb(self, action: str, *args):
        """Handle USB device"""
        if action == 'open':
            device = args[0]  # Format: 'usb:1234:5678'
            try:
                import usb.core
                import usb.util
                
                vendor_id, product_id = map(lambda x: int(x, 16), device.split(':')[1:])
                dev = usb.core.find(idVendor=vendor_id, idProduct=product_id)
                
                if dev is None:
                    print(f"❌ USB device {vendor_id:04x}:{product_id:04x} not found")
                    return None
                
                # Detach kernel driver if needed
                if dev.is_kernel_driver_active(0):
                    dev.detach_kernel_driver(0)
                
                dev.set_configuration()
                return dev
                
            except ImportError:
                print("⚠️  Install pyusb: pip install pyusb")
                return None
        
        elif action == 'send':
            device, data = args
            dev = self.active_interfaces[device]['handle']
            try:
                # USB endpoint 0x01 for OUT (device specific)
                endpoint_out = 0x01
                dev.write(endpoint_out, data)
                return True
            except:
                return False
    
    def _handle_serial(self, action: str, *args):
        """Handle serial port"""
        if action == 'open':
            device = args[0]  # e.g., '/dev/ttyUSB0'
            try:
                import serial
                ser = serial.Serial(
                    port=device,
                    baudrate=115200,
                    bytesize=serial.EIGHTBITS,
                    parity=serial.PARITY_NONE,
                    stopbits=serial.STOPBITS_ONE,
                    timeout=1,
                    xonxoff=False,
                    rtscts=False,
                    dsrdtr=False
                )
                return ser
            except ImportError:
                print("⚠️  Install pyserial: pip install pyserial")
                return None
        
        elif action == 'send':
            device, data = args
            ser = self.active_interfaces[device]['handle']
            return ser.write(data) > 0
        
        elif action == 'receive':
            device, timeout = args
            ser = self.active_interfaces[device]['handle']
            ser.timeout = timeout
            return ser.read(4096)
    
    def _handle_audio(self, action: str, *args):
        """Handle audio interface"""
        if action == 'open':
            device = args[0]  # e.g., 'hw:0,0'
            try:
                import pyaudio
                p = pyaudio.PyAudio()
                
                # Open audio stream
                stream = p.open(
                    format=pyaudio.paInt16,
                    channels=2,
                    rate=44100,
                    output=True,
                    input=True,
                    frames_per_buffer=1024
                )
                return {'pyaudio': p, 'stream': stream}
                
            except ImportError:
                print("⚠️  Install pyaudio: pip install pyaudio")
                return None
        
        elif action == 'send':
            device, data = args
            audio_info = self.active_interfaces[device]['handle']
            stream = audio_info['stream']
            
            # Convert bytes to audio signal
            import numpy as np
            if isinstance(data, bytes):
                # Convert to numpy array for audio
                audio_data = np.frombuffer(data, dtype=np.int16)
                stream.write(audio_data.tobytes())
                return True
            return False
    
    def _handle_magnetic_stripe(self, action: str, *args):
        """Handle magnetic stripe reader/writer"""
        if action == 'open':
            # Magnetic stripe can be connected via:
            # 1. Audio jack (play WAV file)
            # 2. Serial port (specific reader)
            # 3. USB HID (keyboard emulator)
            
            print("\n🔸 Magnetic Stripe Interface Options:")
            print("  1. Audio Jack Output (play .wav file)")
            print("  2. Serial Connection")
            print("  3. USB HID (keyboard emulation)")
            
            choice = input("Select method [1]: ").strip() or "1"
            
            if choice == "1":
                # Use audio interface
                return self._handle_audio('open', 'default')
            elif choice == "2":
                # Use serial
                return self._handle_serial('open', '/dev/ttyUSB0')
            else:
                return None
        
        elif action == 'send':
            device, data = args
            # data should be track1/track2 format
            print(f"📟 Sending magnetic stripe data: {data[:50]}...")
            return True
    
    def _handle_can_bus(self, action: str, *args):
        """Handle CAN bus interface"""
        if action == 'open':
            device = args[0]  # e.g., 'can0'
            try:
                import can
                
                # Create CAN bus interface
                bus = can.Bus(
                    interface='socketcan',
                    channel=device,
                    bitrate=500000
                )
                return bus
                
            except ImportError:
                print("⚠️  Install python-can: pip install python-can")
                return None
        
        elif action == 'send':
            device, data = args
            bus = self.active_interfaces[device]['handle']
            try:
                import can
                msg = can.Message(
                    arbitration_id=0x123,
                    data=data,
                    is_extended_id=False
                )
                bus.send(msg)
                return True
            except:
                return False
    
    def _handle_i2c(self, action: str, *args):
        """Handle I2C interface"""
        if action == 'open':
            device = args[0]  # e.g., 'i2c-1'
            try:
                import smbus
                bus_num = int(device.split('-')[1])
                bus = smbus.SMBus(bus_num)
                return bus
            except ImportError:
                print("⚠️  Install smbus2: pip install smbus2")
                return None
        
        elif action == 'send':
            device, data = args
            bus = self.active_interfaces[device]['handle']
            # I2C write to address 0x50
            try:
                bus.write_i2c_block_data(0x50, 0x00, list(data))
                return True
            except:
                return False
    
    def _handle_spi(self, action: str, *args):
        """Handle SPI interface"""
        if action == 'open':
            device = args[0]  # e.g., 'spidev0.0'
            try:
                import spidev
                spi = spidev.SpiDev()
                bus, device_num = map(int, device.split('.')[0][-1]), int(device.split('.')[1])
                spi.open(bus, device_num)
                spi.max_speed_hz = 1000000
                spi.mode = 0
                return spi
            except ImportError:
                print("⚠️  Install spidev: pip install spidev")
                return None
        
        elif action == 'send':
            device, data = args
            spi = self.active_interfaces[device]['handle']
            response = spi.xfer(list(data))
            return bytes(response) if response else None
    
    def _handle_gpio(self, action: str, *args):
        """Handle GPIO interface"""
        if action == 'open':
            try:
                import RPi.GPIO as GPIO
                GPIO.setmode(GPIO.BCM)
                return GPIO
            except ImportError:
                print("⚠️  Install RPi.GPIO: pip install RPi.GPIO")
                return None
        
        elif action == 'send':
            device, data = args
            gpio = self.active_interfaces[device]['handle']
            # GPIO bit-banging
            for i, bit in enumerate(data):
                gpio.output(17 + i, bit)
            return True

# ===== UNIVERSAL PROTOCOL BRIDGE =====

class PhantomProtocolBridge:
    """
    Bridge between ANY protocols and interfaces
    Example: Ethernet -> USB, Bluetooth -> Serial, etc.
    """
    
    def __init__(self):
        self.controller = PhantomInterfaceController()
        self.bridges = {}
        self.running = False
    
    def create_bridge(self, src_interface: str, dst_interface: str):
        """Create bridge between two interfaces"""
        print(f"\n🔗 Creating bridge: {src_interface} → {dst_interface}")
        
        # Open both interfaces
        src_opened = False
        dst_opened = False
        
        # Try to determine interface types
        for iface_type in InterfaceType:
            if iface_type.value in src_interface.lower():
                src_opened = self.controller.open_interface(iface_type, src_interface)
                break
        
        for iface_type in InterfaceType:
            if iface_type.value in dst_interface.lower():
                dst_opened = self.controller.open_interface(iface_type, dst_interface)
                break
        
        if not src_opened or not dst_opened:
            print("❌ Failed to open interfaces for bridge")
            return False
        
        bridge_id = f"{src_interface}_to_{dst_interface}"
        
        def bridge_thread():
            """Thread that forwards data between interfaces"""
            print(f"⚡ Bridge {bridge_id} active")
            
            while self.running:
                try:
                    # Receive from source
                    data = self.controller.receive_data(src_interface, 0.1)
                    if data:
                        # Process/transform if needed
                        transformed = self.transform_data(data, src_interface, dst_interface)
                        
                        # Send to destination
                        self.controller.send_data(dst_interface, transformed)
                        
                        print(f"  📦 {len(data)} bytes: {src_interface} → {dst_interface}")
                
                except Exception as e:
                    print(f"⚠️  Bridge error: {e}")
                    time.sleep(1)
        
        # Start bridge thread
        thread = threading.Thread(target=bridge_thread, daemon=True)
        thread.start()
        
        self.bridges[bridge_id] = {
            'thread': thread,
            'src': src_interface,
            'dst': dst_interface,
            'started': time.time()
        }
        
        return True
    
    def transform_data(self, data: bytes, src: str, dst: str) -> bytes:
        """Transform data between different protocols"""
        # Simple transformations based on interface types
        
        # Ethernet to USB
        if 'eth' in src and 'usb' in dst:
            # Strip Ethernet header, add USB header
            if len(data) > 14:  # Ethernet header
                return b'\x01' + data[14:]  # Simple USB packet
        
        # Serial to Ethernet
        if 'tty' in src and 'eth' in dst:
            # Wrap in Ethernet frame
            eth_header = binascii.unhexlify('ffffffffffff0000000000000800')
            return eth_header + data
        
        # Audio to Serial (magnetic stripe simulation)
        if 'audio' in src and 'tty' in dst:
            # Convert audio to serial bits
            # This is a simplified version
            bits = ''.join(['1' if b > 128 else '0' for b in data[:100]])
            return bits.encode()
        
        return data  # No transformation
    
    def list_bridges(self):
        """List active bridges"""
        print("\n" + "="*80)
        print("🌉 ACTIVE PROTOCOL BRIDGES")
        print("="*80)
        
        if not self.bridges:
            print("  No active bridges")
            return
        
        for bridge_id, info in self.bridges.items():
            uptime = time.time() - info['started']
            print(f"\n  🔗 {bridge_id}")
            print(f"     Source: {info['src']}")
            print(f"     Destination: {info['dst']}")
            print(f"     Uptime: {uptime:.1f}s")
        
        print("\n" + "="*80)
    
    def start(self):
        """Start bridge controller"""
        self.running = True
        print("🚀 Phantom Protocol Bridge started")
    
    def stop(self):
        """Stop all bridges"""
        self.running = False
        for bridge_id, info in self.bridges.items():
            print(f"🛑 Stopping bridge: {bridge_id}")
        self.bridges.clear()

class UniversalMagneticStripeInjector:
    """
    Inject magnetic stripe data through ANY interface
    """
    
    def __init__(self):
        self.controller = PhantomInterfaceController()
        self.encoder = MagneticStripeEncoder()
        
    def inject_via_interface(self, interface_type: InterfaceType, 
                           device: str, track1: str, track2: str):
        """
        Inject magnetic stripe data through specified interface
        
        Supports:
        - Audio jack (play WAV)
        - Serial port (direct to reader)
        - USB HID (keyboard emulation)
        - Ethernet (packet injection)
        - Bluetooth (RFCOMM)
        """
        
        print(f"\n💉 Injecting magnetic stripe via {interface_type.value}")
        print(f"   Device: {device}")
        print(f"   Track 1: {track1[:30]}...")
        print(f"   Track 2: {track2[:30]}...")
        
        if interface_type == InterfaceType.AUDIO:
            # Generate audio and play through audio jack
            return self._inject_via_audio(device, track1, track2)
        
        elif interface_type == InterfaceType.SERIAL:
            # Send directly to serial port (for USB magnetic readers)
            return self._inject_via_serial(device, track1, track2)
        
        elif interface_type == InterfaceType.USB:
            # USB HID keyboard emulation
            return self._inject_via_usb_hid(device, track1, track2)
        
        elif interface_type == InterfaceType.ETHERNET:
            # Encapsulate in network packets
            return self._inject_via_ethernet(device, track1, track2)
        
        elif interface_type == InterfaceType.BLUETOOTH:
            # Send via Bluetooth
            return self._inject_via_bluetooth(device, track1, track2)
        
        else:
            print(f"❌ Interface {interface_type} not supported for magnetic stripe")

    def _bits_to_fsk(self, bits: str, sample_rate: int, duration: float,
                    freq_0: int = 1200, freq_1: int = 2400, baud: int = 1200) -> np.ndarray:
        """Convert bits to FSK audio signal"""
        import numpy as np
        
        samples_per_bit = sample_rate // baud
        total_samples = int(sample_rate * duration)
        
        audio = np.zeros(total_samples)
        
        for i, bit in enumerate(bits):
            if i * samples_per_bit >= total_samples:
                break
            
            start = i * samples_per_bit
            end = min((i + 1) * samples_per_bit, total_samples)
            length = end - start
            
            freq = freq_0 if bit == '0' else freq_1
            t = np.linspace(0, length / sample_rate, length, False)
            tone = 0.8 * np.sin(2 * np.pi * freq * t)
            
            audio[start:end] = tone
        
        return audio
